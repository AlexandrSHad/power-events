## 2.5 Extend Animation Timer

Add to `update_metrics_animation()`:

```cpp
// Smooth interpolation for temperatures (faster factor)
current_cpu_temp += (target_cpu_temp - current_cpu_temp) * 0.3f;
current_gpu_temp += (target_gpu_temp - current_gpu_temp) * 0.3f;

// Smooth interpolation for battery (slower factor)
current_battery_percent += (target_battery_percent - current_battery_percent) * 0.1f;

// Update temperature labels
if (cpu_temp_label != NULL) {
    char buf[16];
    snprintf(buf, sizeof(buf), target_cpu_temp > 0 ? "%.0f°" : "--°", current_cpu_temp);
    lv_label_set_text(cpu_temp_label, buf);
}
if (gpu_temp_label != NULL) {
    char buf[16];
    snprintf(buf, sizeof(buf), target_gpu_temp > 0 ? "%.0f°" : "--°", current_gpu_temp);
    lv_label_set_text(gpu_temp_label, buf);
}

// Update battery fill and label
if (battery_fill_obj != NULL && battery_data_received) {
    int fill_width = constrain((int)(current_battery_percent / 100.0f * 26.0f), 1, 26);
    lv_obj_set_width(battery_fill_obj, fill_width);

    // Color: green (>20% or charging), amber (10-20%), red (<10%)
    if (battery_charging || current_battery_percent >= 20) {
        lv_obj_set_style_bg_color(battery_fill_obj, COLOR_CHECKMARK, 0);
    } else if (current_battery_percent >= 10) {
        lv_obj_set_style_bg_color(battery_fill_obj, COLOR_QUESTION_MARK, 0);
    } else {
        lv_obj_set_style_bg_color(battery_fill_obj, lv_color_hex(0xFF1744), 0);
    }
}

if (battery_pct_label != NULL && battery_data_received) {
    char buf[16];
    snprintf(buf, sizeof(buf), "%.0f%%", current_battery_percent);
    lv_label_set_text(battery_pct_label, buf);
}

// Show/hide lightning bolt charging icon
if (battery_charging_icon != NULL && battery_data_received) {
    if (battery_charging) {
        lv_obj_clear_flag(battery_charging_icon, LV_OBJ_FLAG_HIDDEN);
    } else {
        lv_obj_add_flag(battery_charging_icon, LV_OBJ_FLAG_HIDDEN);
    }
}
```
